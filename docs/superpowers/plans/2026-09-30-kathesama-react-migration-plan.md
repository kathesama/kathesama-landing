# Kathesama React Migration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Migrate the existing single-file Kathesama landing page to a tested React and TypeScript application without changing its published content or recognizable visual identity.

**Architecture:** A Vite application renders a small React Router page tree inside a shared shell. Language state is derived from `?lang=en|es`, persisted locally, and supplied through a focused context. Existing landing copy is moved into typed bilingual content records and composed from small section components; the architecture and blog routes remain navigation-ready placeholders until their separate plans are executed.

**Tech Stack:** React, TypeScript, Vite, React Router, Vitest, React Testing Library, jest-dom, user-event, axe-core, CSS

---

Create `src/features/i18n/language.ts`:

```ts
export const supportedLanguages = ['en', 'es'] as const;
export type Language = (typeof supportedLanguages)[number];

export const isLanguage = (value: string | null): value is Language =>
  value === 'en' || value === 'es';

export function resolveLanguage(
  search: string,
  storedLanguage: string | null,
  browserLanguage: string,
): Language {
  const queryLanguage = new URLSearchParams(search).get('lang');
  if (isLanguage(queryLanguage)) return queryLanguage;
  if (isLanguage(storedLanguage)) return storedLanguage;
  if (browserLanguage.toLowerCase().startsWith('es')) return 'es';
  return 'en';
}

export function setLanguageInSearch(search: string, language: Language): string {
  const params = new URLSearchParams(search);
  params.set('lang', language);
  return `?${params.toString()}`;
}
```

```tsx
import { createContext, useContext, useEffect, useMemo, type PropsWithChildren } from 'react';
import { useLocation, useNavigate } from 'react-router-dom';
import { resolveLanguage, setLanguageInSearch, type Language } from './language';

type LanguageContextValue = {
  language: Language;
  setLanguage: (language: Language) => void;
};

const LanguageContext = createContext<LanguageContextValue | null>(null);
const storageKey = 'kathesama-language';

export function LanguageProvider({ children }: PropsWithChildren) {
  const location = useLocation();
  const navigate = useNavigate();
  const language = resolveLanguage(
    location.search,
    window.localStorage.getItem(storageKey),
    window.navigator.language,
  );

  useEffect(() => {
    document.documentElement.lang = language;
    window.localStorage.setItem(storageKey, language);
  }, [language]);

  const value = useMemo<LanguageContextValue>(() => ({
    language,
    setLanguage(nextLanguage) {
      navigate({
        pathname: location.pathname,
        search: setLanguageInSearch(location.search, nextLanguage),
        hash: location.hash,
      });
    },
  }), [language, location.hash, location.pathname, location.search, navigate]);

  return <LanguageContext.Provider value={value}>{children}</LanguageContext.Provider>;
}

export function useLanguage(): LanguageContextValue {
  const value = useContext(LanguageContext);
  if (!value) throw new Error('useLanguage must be used inside LanguageProvider');
  return value;
}
```

```tsx
import { useEffect, useRef } from 'react';

export function AnimatedGrid() {
  const canvasRef = useRef<HTMLCanvasElement>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    const context = canvas?.getContext('2d');
    if (!canvas || !context) return;
    const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    let frame = 0;
    let animationId = 0;

    const renderFrame = () => {
      context.clearRect(0, 0, canvas.width, canvas.height);
      for (let x = 0; x <= canvas.width; x += 60) {
        for (let y = 0; y <= canvas.height; y += 60) {
          const alpha = reduceMotion ? 0.13 : 0.08 + 0.05 * Math.sin(frame / 45 + (x + y) / 80);
          context.beginPath();
          context.arc(x, y, 1.2, 0, Math.PI * 2);
          context.fillStyle = `rgba(0, 229, 192, ${alpha})`;
          context.fill();
        }
      }
    };

    const tick = () => {
      renderFrame();
      frame += 1;
      animationId = window.requestAnimationFrame(tick);
    };

    const resize = () => {
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;
      renderFrame();
    };

    resize();
    if (!reduceMotion) animationId = window.requestAnimationFrame(tick);
    window.addEventListener('resize', resize);
    return () => {
      window.removeEventListener('resize', resize);
      window.cancelAnimationFrame(animationId);
    };
  }, []);

  return <canvas ref={canvasRef} id="canvas" aria-hidden="true" />;
}
```
